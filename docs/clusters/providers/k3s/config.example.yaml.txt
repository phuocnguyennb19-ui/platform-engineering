# Reference K3s server configuration — copy to /etc/rancher/k3s/config.yaml on the first
# server BEFORE installing. Named *.txt so no tool in this repository reads it as YAML or
# as a manifest. Every value is a choice to review, not a requirement.
#
# Never put `token:` here or anywhere in git: the join token is generated on the server
# (/var/lib/rancher/k3s/server/node-token) and moved over a secure channel.

# API certificate must cover the address clients use (VIP, load balancer or DNS name).
tls-san:
  - <api-address>

# The repository's values expect the Kong IngressClass (kong); the bundled
# Traefik would register a second, unused class. Keep ServiceLB unless another load
# balancer (MetalLB, kube-vip) assigns LoadBalancer addresses.
disable:
  - traefik

# Kubeconfig on the server stays root-only.
write-kubeconfig-mode: "0600"
